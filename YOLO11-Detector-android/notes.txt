nano run_app.sh

#!/bin/bash
# Move to project directory
cd "/mnt/c/Users/User/Downloads/flet works/YOLO11-Detector"

# Activate working virtual environment
source .venv/bin/activate

# Force software rendering (prevents the libEGL / MESA warnings)
export LIBGL_ALWAYS_SOFTWARE=1

# Run the app
flet run main.py

Press Ctrl+O, Enter, then Ctrl+X to save and exit.

chmod +x run_app.sh
